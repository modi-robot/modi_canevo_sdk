完成配置后，执行下面指令：

```
# 编译内核(需要点时间)
cd /path/to/linux-6.8.2
make -j$(nproc)

# 安装模块
sudo make INSTALL_MOD_STRIP=1 modules_install

# 安装内核（跳过自动生成 initramfs）
sudo make install INSTALL_MOD_STRIP=1

# 手动生成精简的 initramfs
# 修改配置
sudo sed -i 's/^MODULES=.*/MODULES=dep/' /etc/initramfs-tools/initramfs.conf
sudo sed -i 's/^COMPRESS=.*/COMPRESS=xz/' /etc/initramfs-tools/initramfs.conf

# 生成initramfs
sudo mkinitramfs -o /boot/initrd.img-6.8.2-rt11 6.8.2-rt11

# 更新引导
sudo update-grub
```

```
# 查看编译后核心，注意核心大小
cd /boot
ls -l
```

### 3、配置隔离的cpu，以下是对隔离CPU2的示例，用户可以自行选择需要隔离的CPU

```
# 打开配置页面
sudo nano /etc/default/grub

# 将GRUB_CMDLINE_LINUX_DEFAULT那一行改为

GRUB_CMDLINE_LINUX_DEFAULT="quiet splash \
isolcpus=2 nohz_full=2 rcu_nocbs=2 rcu_nocb_poll \
nohz=on nosmt \
irqaffinity=0,1,3 \
rcupdate.rcu_expedited=1 \
kernel.nmi_watchdog=0 \
audit=0 \
mce=off \
idle=poll \
nosoftlockup \
page_alloc.shuffle=0"

# 
sudo nano /etc/systemd/system.conf
把CPUAffinity = 改成CPUAffinity =0 1 3
# 
sudo nano /etc/systemd/user.conf
在最后加上CPUAffinity =0 1 3

# 更新并重启
sudo update-grub
reboot
```

### 5、配置CAN-FD

```
sudo ip link set can0 type can bitrate 1000000 dbitrate 5000000 fd on
sudo ip link set can0 up

# 设置运行的进程不要跑到cpu2上
sudo ps -eo pid | awk 'NR>1' | xargs -I {} sudo taskset -pc 0,1,3 {} 2>/dev/null
sudo find /proc/irq -maxdepth 1 -type d -name "[0-9]*" | sort -n | xargs -I {} sudo sh -c "echo 0,1,3 > {}/smp_affinity_list" 2>/dev/null
```
