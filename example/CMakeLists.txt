  cmake_minimum_required(VERSION 3.14)
  project(modi_canevo_sdk_examples LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

find_package(Threads REQUIRED)

if(CANEVO_EXAMPLE_SOURCE_BUILD OR TARGET modi_canevo_sdk)
  set(CANEVO_EXAMPLE_LINK_TARGET modi_canevo_sdk)
else()
  set(SDK_ROOT_DIR "${CMAKE_CURRENT_LIST_DIR}/..")
  list(PREPEND CMAKE_PREFIX_PATH "${SDK_ROOT_DIR}")
  # 发布包与示例相邻，交叉编译时也直接查找该目录，不追加 sysroot 前缀。
  find_package(modi_canevo_sdk CONFIG REQUIRED NO_CMAKE_FIND_ROOT_PATH)
  if(NOT TARGET modi_canevo_sdk::modi_canevo_sdk)
    message(FATAL_ERROR "Target not found: modi_canevo_sdk::modi_canevo_sdk")
  endif()
  set(CANEVO_EXAMPLE_LINK_TARGET modi_canevo_sdk::modi_canevo_sdk)
endif()

function(add_sdk_example target_name source_file)
  add_executable(${target_name} ${source_file})
  target_compile_definitions(${target_name} PRIVATE _GNU_SOURCE)
  target_link_libraries(${target_name} PRIVATE ${CANEVO_EXAMPLE_LINK_TARGET} Threads::Threads)
  if(TARGET modi_canevo_sdk)
    set_target_properties(${target_name} PROPERTIES
      INSTALL_RPATH "$ORIGIN/../lib;$ORIGIN/../lib/runtime")
    install(TARGETS ${target_name} RUNTIME DESTINATION bin COMPONENT LocalDeployment)
  endif()
endfunction()
