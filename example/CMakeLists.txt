if(CMAKE_CURRENT_SOURCE_DIR STREQUAL CMAKE_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.14)
  project(canevo_sdk_x86_examples LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()

if(CANEVO_EXAMPLE_SOURCE_BUILD OR TARGET canevo_sdk_x86)
  set(CANEVO_EXAMPLE_LINK_TARGET canevo_sdk_x86)
else()
  set(SDK_ROOT_DIR "${CMAKE_CURRENT_LIST_DIR}/..")
  list(PREPEND CMAKE_PREFIX_PATH "${SDK_ROOT_DIR}")
  find_package(canevo_sdk_x86 CONFIG REQUIRED)
  if(NOT TARGET canevo_sdk_x86::canevo_sdk_x86)
    message(FATAL_ERROR "Target not found: canevo_sdk_x86::canevo_sdk_x86")
  endif()
  set(CANEVO_EXAMPLE_LINK_TARGET canevo_sdk_x86::canevo_sdk_x86)
endif()

set(CANEVO_HAVE_NECRO OFF)
find_package(Necro QUIET)
if(Necro_FOUND AND TARGET Necro::qiuniu)
  set(CANEVO_HAVE_NECRO ON)
  message(STATUS "canevo examples: NECRO found - using Necro::qiuniu for RT examples")
else()
  message(STATUS "canevo examples: NECRO not found - RT examples use POSIX fallback")
endif()

function(add_sdk_example target_name source_file)
  add_executable(${target_name} ${source_file})
  target_link_libraries(${target_name} PRIVATE ${CANEVO_EXAMPLE_LINK_TARGET})
endfunction()

function(enable_necro_rt target_name)
  target_compile_definitions(${target_name} PRIVATE _GNU_SOURCE)
  if(CANEVO_HAVE_NECRO)
    target_compile_definitions(${target_name} PRIVATE CANEVO_HAVE_NECRO=1 __QIUNIU__)
    target_compile_options(${target_name} PRIVATE -fasynchronous-unwind-tables)
    target_link_libraries(${target_name} PRIVATE Necro::qiuniu)
    if(COMMAND necro_target_bootstrap)
      necro_target_bootstrap(${target_name})
    endif()
  else()
    target_compile_definitions(${target_name} PRIVATE CANEVO_HAVE_NECRO=0)
  endif()
endfunction()
